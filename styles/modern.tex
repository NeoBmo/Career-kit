% ---------- Encabezado / identidad ----------
\newcommand{\ckHeader}[3]{%
  % #1 = nombre, #2 = titulo profesional, #3 = linea de contacto
  {\headingfont\bfseries\fontsize{25}{27}\selectfont #1}\par
  \vspace{1pt}
  {\headingfont\color{ckAccent}\large #2}\par
  \vspace{3pt}
  {\small\color{ckSubtle} #3}\par
  \vspace{4pt}
  {\color{ckRule}\rule{\linewidth}{1.2pt}}
}

% ---------- Fila experiencia/educación: título + fecha alineada a la derecha ----------
\newcommand{\ckEntry}[4]{%
  % #1 = titulo del cargo/programa, #2 = fecha, #3 = organizacion, #4 = subtitulo opcional
  \noindent
  \begin{tabularx}{\linewidth}{@{}X r@{}}
    {\headingfont\bfseries #1} & {\small\color{ckSubtle} #2}
  \end{tabularx}\\[-1pt]
  {\itshape\color{ckSubtle} #3}%
  \ifx&#4&\else\ ---\ \itshape #4\fi\par
}
